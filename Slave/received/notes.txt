function cipher_txt = RC4(plain_txt,key)
s=zeros(1,256);
t=zeros(1,256);
key=double(key);
for i=0:255
    s(i+1)=i;
    t(i+1)=key(mod(i,lengt(key))+1);
end
j=0;
for i=0:255
    j=mod(j+s(i+1)+t(i+1),256);
    [s(i+1),s(j+1)]=deal(s(j+1),s(i+1));
end
i=0;j=0;
for l=1:length(plain_txt)
    i=mod(i+1,256);
    j=mod(j+s(i+1),256);
    [s(i+1),s(j+1)]=deal(s(j+1),s(i+1));
    a=s(mod(s(i+1)+s(j+1),256)+1);
    cipher_txt(l)=char(bitxor(doule(plain_txt(l)),a));
end
end
